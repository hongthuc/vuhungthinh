        /* Điều khiển 2 Banner Tự Trượt Chuyên Nghiệp */
        let currentBannerIndex = 0;
        const totalBanners = 2;
        let bannerInterval = null;

        function showBanner(index) {
            currentBannerIndex = (index + totalBanners) % totalBanners;
            for (let i = 0; i < totalBanners; i++) {
                const slide = document.getElementById('banner-slide-' + i);
                const dot = document.getElementById('banner-dot-' + i);
                if (i === currentBannerIndex) {
                    slide.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
                    slide.classList.add('opacity-100', 'z-10');
                    dot.className = 'w-6 h-2 rounded-full bg-vht-amber transition-all';
                } else {
                    slide.classList.remove('opacity-100', 'z-10');
                    slide.classList.add('opacity-0', 'z-0', 'pointer-events-none');
                    dot.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
                }
            }
        }

        function changeSlide(step) {
            showBanner(currentBannerIndex + step);
            resetBannerTimer();
        }

        function goToSlide(index) {
            showBanner(index);
            resetBannerTimer();
        }

        function resetBannerTimer() {
            if (bannerInterval) clearInterval(bannerInterval);
            bannerInterval = setInterval(() => {
                showBanner(currentBannerIndex + 1);
            }, 5500);
        }

        /* Khởi chạy khi tải trang */
        window.onload = function () {
            renderConstructionItems('all');
            renderDaNangProjects('all');
            resetBannerTimer();
        };
